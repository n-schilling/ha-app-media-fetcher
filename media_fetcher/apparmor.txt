#include <tunables/global>

# Fetches media from the internet with yt-dlp, Deno and ffmpeg and writes
# only its job files to /data
profile media_fetcher flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  capability chown,
  capability dac_override,
  capability fowner,
  capability kill,
  capability setgid,
  capability setuid,

  network inet,
  network inet6,
  network unix,

  # s6 stops its services; the Supervisor stops the container
  signal (send, receive),

  file,
  deny /proc/sys/** w,
  deny /sys/** w,

  # ffmpeg and ffprobe convert files that are already downloaded, and Deno
  # solves the JavaScript a site hands to yt-dlp: none of them needs the
  # network. Without it, a download that names other addresses (a manifest
  # disguised as media) leads nowhere, whatever ffmpeg makes of it.
  /usr/bin/ffmpeg Cx -> offline,
  /usr/bin/ffprobe Cx -> offline,
  /opt/venv/bin/deno Cx -> offline,

  profile offline flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    network unix,
    deny network inet,
    deny network inet6,
    signal (send, receive),

    file,
    deny /proc/sys/** w,
    deny /sys/** w,
  }
}
