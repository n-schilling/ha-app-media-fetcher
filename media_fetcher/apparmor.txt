abi <abi/3.0>,
#include <tunables/global>

# The ABI line makes kernels with fine-grained network mediation enforce the
# network rules per profile; without it they fall back to a compatibility
# mode in which the child profile below would keep the network

# Fetches media from the internet with yt-dlp, Deno and ffmpeg and writes
# only its job files to /data. s6-overlay runs as root and hands /data to
# user fetcher, as which the service runs. Measured on Home Assistant OS in
# complain mode before it enforced.
profile media_fetcher flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # s6-overlay, and the hand-over of /data to the service's user
  capability chown,
  capability dac_override,
  capability fowner,
  capability kill,
  capability setgid,
  capability setuid,

  network inet dgram,
  network inet6 dgram,
  network inet stream,
  network inet6 stream,
  network unix,

  # s6 stops its services; the Supervisor stops the container
  signal (send, receive),

  # s6-overlay
  /init rix,
  /package/** rix,
  /command/** rix,
  /etc/s6-overlay/** rix,
  /run/** rwk,
  /run/ r,

  # Programs and libraries of the image
  /bin/** rix,
  /sbin/** rix,
  /usr/bin/** rix,
  /usr/sbin/** rix,
  /usr/local/bin/** rix,
  /usr/lib/** rm,
  /usr/local/lib/** rm,
  /lib/** rm,
  /opt/venv/** rm,
  /opt/venv/bin/* rix,
  /usr/share/** r,
  /etc/** r,

  # ffmpeg and ffprobe convert files that are already downloaded, and Deno
  # solves the JavaScript a site hands to yt-dlp: none of them needs the
  # network. Without it, a download that names other addresses (a manifest
  # disguised as media) leads nowhere, whatever ffmpeg makes of it.
  /usr/bin/ffmpeg Cx -> offline,
  /usr/bin/ffprobe Cx -> offline,
  /opt/venv/bin/deno Cx -> offline,

  # bashio; temporary files of yt-dlp
  /tmp/ r,
  /tmp/** rwk,

  # Options and jobs
  /data/ r,
  /data/** rwk,

  @{PROC}/@{pid}/** r,
  @{PROC}/ r,
  @{PROC}/sys/kernel/** r,
  @{PROC}/cpuinfo r,
  @{PROC}/meminfo r,
  /sys/devices/system/cpu/** r,
  /sys/fs/cgroup/** r,
  /dev/null rw,
  /dev/urandom r,
  /dev/random r,
  /dev/tty rw,

  deny /proc/sys/** w,
  deny /sys/** w,

  profile offline flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    network unix,
    signal (send, receive),

    /usr/bin/ffmpeg rm,
    /usr/bin/ffprobe rm,
    /opt/venv/bin/deno rm,
    /usr/lib/** rm,
    /lib/** rm,
    /opt/venv/** r,
    /usr/share/** r,
    /etc/** r,

    # The job's files
    /data/jobs/** rwk,
    /tmp/** rwk,

    @{PROC}/@{pid}/** r,
    @{PROC}/sys/kernel/** r,
    @{PROC}/cpuinfo r,
    @{PROC}/meminfo r,
    /sys/devices/system/cpu/** r,
    /sys/fs/cgroup/** r,
    /dev/null rw,
    /dev/urandom r,
    /dev/random r,

    deny /proc/sys/** w,
    deny /sys/** w,
  }
}
